
// before C++11
struct Type {
private:
    Resource *m_res_p;
public:
    // ...

    Type& move_out(void *mem) noexcept { // move without Type&&; manual placement new
        Type *new_type_p = static_cast<Type*>(mem);
        new_type_p->m_res_p = this->m_res_p;
        this->m_res_p       = NULL;
        return *new_type_p;
    }
};

// caller side:
Type obj;
unsigned char buf[sizeof(Type)];
Type& obj_moved_ref = obj.move_out(buf);

// after C++11

struct Type {
private:
    Resource *m_res_p;
public:
    // ...

    Type(Type&& other_rref) noexcept
        m_res_p(other_rref.m_res_p)
    { // move with Type&&; std::move
        other_rref.m_res_p = nullptr;
    }

    Type& operator=(Type&& other_rref) noexcept { // move with Type&&; std::move
        if (this != &other) {
            delete m_res_p;
            this->m_res_p = other_rref.m_res_p;
            other_rref.m_res_p = nullptr;
            return *this;
        }
    }
};

// caller side:
Type obj;
...
Type obj1(std::move(obj));
auto obj2 = std::move(obj1);


// perfect language syntax

import std.print
import std.vector

class Point {
public:
    int m_x;
    int m_y;

    Point(int x, y) {
        this.m_x = x;
        this.m_y = y;
    }

    void print() {
        std.println("{}, {}", x, y);
    }

    void add(int dx, int dy) {
        this.m_x += dx;
        this.m_y += dy;
    }

    

    ~Point() {
        this.m_x = 0;
        this.m_y = 0;
    }
};

Point add_points(Point p1, Point p2) {
    Point p = new Point(p1.m_x, p1.m_y);
    p.add(p2.m_x, p2.m_y);
}

int main() {
    Point p = new Point(2, 3);

    p.print();
    p.add(3, 4);
    p.print();

    std.vector<int> v = new vector();
    v.push(1);
    v.push(2);




    // v, p automatically freed
    return 0;
}
